// Фасад сценария внутри песочницы (#448). Наружу — один мост __bridge без прототипа; туда и
// обратно идут только строки JSON, так что объекты хоста в песочницу не попадают.
const __call = (op, input) =>
  new Promise((ok, fail) =>
    __bridge(op, JSON.stringify(input === undefined ? null : input), (text) => ok(text === undefined ? undefined : JSON.parse(text)), (message) => fail(new Error(message))),
  );
const __sync = (op, input) => {
  let out;
  __bridge(op, JSON.stringify(input === undefined ? null : input), (text) => { out = text === undefined ? undefined : JSON.parse(text) }, (message) => { throw new Error(message) });
  return out;
};
let __anonymous = 0;
function agent(name, persona) {
  const named = typeof name === "string" && name.length > 0;
  const key = named ? name : "#" + ++__anonymous;
  const role = typeof persona === "string" ? persona : persona && typeof persona.system === "string" ? persona.system : undefined;
  __sync("agent", { key, name: named ? name : undefined, role });
  return Object.freeze({ ask: (instructions, schema) => __call("ask", { key, instructions: String(instructions), schema: typeof schema === "number" ? schema : null }) });
}
